```rust
//! DMA bump allocator and VirtIO Hal implementation.
//!
//! Uses a fixed region at 0x8200_0000 for DMA buffers.
//! The kernel identity-maps physical RAM, so phys_to_virt is identity.

use core::sync::atomic::{AtomicUsize, Ordering};
use virtio_drivers::{Hal, PhysAddr, VirtAddr};

/// Fixed base address for the DMA pool — above kernel heap (24 MiB ends at 0x8180_0000).
const POOL_BASE: usize = 0x8200_0000;

/// 5 MiB pool.
const POOL_SIZE: usize = 5 * 1024 * 1024;

const PAGE_SIZE: usize = 4096;

/// Atomic offset into the pool.
static NEXT: AtomicUsize = AtomicUsize::new(0);

/// Bump-allocate `size` bytes with the given alignment from the pool.
fn bump_alloc(size: usize, align: usize) -> *mut u8 {
    loop {
        let current = NEXT.load(Ordering::Relaxed);
        let addr = POOL_BASE + current;
        let aligned = (addr + align - 1) & !(align - 1);
        let offset = aligned - POOL_BASE;
        let new_offset = offset + size;
        if new_offset > POOL_SIZE {
            return core::ptr::null_mut();
        }
        if NEXT
            .compare_exchange(current, new_offset, Ordering::Relaxed, Ordering::Relaxed)
            .is_ok()
        {
            unsafe { core::ptr::write_bytes(aligned as *mut u8, 0, size) };
            return aligned as *mut u8;
        }
    }
}

/// VirtIO HAL: identity-mapped, DMA from fixed pool.
pub struct HalImpl;

impl Hal for HalImpl {
    fn dma_alloc(pages: usize) -> PhysAddr {
        let size = pages * PAGE_SIZE;
        let ptr = bump_alloc(size, PAGE_SIZE);
        if ptr.is_null() { 0 } else { ptr as PhysAddr }
    }

    fn dma_dealloc(_paddr: PhysAddr, _pages: usize) -> i32 {
        0
    }

    fn phys_to_virt(paddr: PhysAddr) -> VirtAddr {
        paddr
    }

    fn virt_to_phys(vaddr: VirtAddr) -> PhysAddr {
        vaddr
    }
}
```

```rust
//! VirtIO device initialization: GPU framebuffer and keyboard input.

use crate::allocator::HalImpl;
use core::ptr::NonNull;
use virtio_drivers::{DeviceType, MmioTransport, Transport, VirtIOGpu, VirtIOHeader, VirtIOInput};

const VIRTIO_MMIO_BASE: usize = 0x10001000;
const VIRTIO_MMIO_END: usize = 0x10008000;
const VIRTIO_MMIO_STRIDE: usize = 0x1000;

/// Framebuffer descriptor returned by GPU initialization.
pub struct Framebuffer {
    /// Raw pointer to the pixel buffer (BGRA format, 4 bytes per pixel).
    pub ptr: *mut u8,
    /// Length of the pixel buffer in bytes.
    pub len: usize,
    /// Screen width in pixels.
    pub width: u32,
    /// Screen height in pixels.
    pub height: u32,
}

/// Result of scanning all VirtIO MMIO slots.
pub struct VirtIODevices {
    /// GPU driver and framebuffer info.
    pub gpu: (VirtIOGpu<'static, HalImpl, MmioTransport>, Framebuffer),
    /// Keyboard input driver (None if no keyboard device found).
    pub keyboard: Option<VirtIOInput<HalImpl, MmioTransport>>,
}

/// Scan all VirtIO MMIO slots, initialize GPU and keyboard devices.
pub fn init() -> VirtIODevices {
    let mut gpu_transport: Option<MmioTransport> = None;
    let mut kbd_transport: Option<MmioTransport> = None;

    let mut addr = VIRTIO_MMIO_BASE;
    while addr <= VIRTIO_MMIO_END {
        let header = NonNull::new(addr as *mut VirtIOHeader).unwrap();
        if let Ok(transport) = unsafe { MmioTransport::new(header) } {
            match transport.device_type() {
                DeviceType::GPU => {
                    tg_console::log::info!("found VirtIO GPU at {addr:#x}");
                    gpu_transport = Some(transport);
                }
                DeviceType::Input => {
                    tg_console::log::info!("found VirtIO Input at {addr:#x}");
                    kbd_transport = Some(transport);
                }
                other => {
                    tg_console::log::debug!("skipping VirtIO device type {other:?} at {addr:#x}");
                }
            }
        }
        addr += VIRTIO_MMIO_STRIDE;
    }

    let gpu_transport = gpu_transport.expect("no VirtIO GPU device found on any MMIO slot");
    let mut gpu = VirtIOGpu::new(gpu_transport).expect("failed to create VirtIOGpu");

    let (width, height) = gpu.resolution().expect("failed to get resolution");
    let (ptr, len) = {
        let buf = gpu.setup_framebuffer().expect("failed to setup framebuffer");
        (buf.as_mut_ptr(), buf.len())
    };

    let fb = Framebuffer {
        ptr,
        len,
        width,
        height,
    };

    let keyboard = kbd_transport.map(|t| {
        VirtIOInput::new(t).expect("failed to create VirtIOInput")
    });

    if keyboard.is_none() {
        tg_console::log::warn!("no VirtIO keyboard device found — input will be unavailable");
    }

    VirtIODevices {
        gpu: (gpu, fb),
        keyboard,
    }
}
```

```rust
/// Translate a VirtIO keycode to an ASCII byte.
fn translate_keycode(code: u16) -> u8 {
    match code {
        17 | 103 => b'w',   // W or Up
        30 | 105 => b'a',   // A or Left
        31 | 108 => b's',   // S or Down
        32 | 106 => b'd',   // D or Right
        28 => b'\r',         // Enter
        57 => b' ',          // Space
        _ => b'\x01',        // Any other key
    }
}

/// Poll VirtIO keyboard for the next key press byte.
#[cfg(target_arch = "riscv64")]
fn keyboard_trygetchar() -> Option<u8> {
    let kbd = unsafe {
        let p = &raw mut KEYBOARD;
        (*p).as_mut()
    }?;
    loop {
        match kbd.pop_pending_event() {
            Some(event) => {
                if event.event_type == 1 && event.value == 1 {
                    return Some(translate_keycode(event.code));
                }
            }
            None => return None,
        }
    }
}
```

- [ ] **Step 5: Add FB_INFO and FB_WRITE handlers**

```rust
/// FB_INFO: returns (width << 32) | height.
#[cfg(target_arch = "riscv64")]
fn handle_fb_info() -> usize {
    let (width, height) = unsafe {
        let p = &raw const FRAMEBUFFER;
        let fb = (*p).as_ref().expect("GPU not initialized");
        (fb.width, fb.height)
    };
    ((width as usize) << 32) | (height as usize)
}

/// FB_WRITE: copy BGRA pixels from user buffer into framebuffer and flush.
///
/// Unlike ch3, the user data pointer must be translated through the process's
/// page table before the kernel can read it.
#[cfg(target_arch = "riscv64")]
fn handle_fb_write(caller_entity: usize, x: usize, y: usize, w: usize, h: usize, data_ptr: usize) -> usize {
    let (fb_ptr, fb_len, fb_w, fb_h) = unsafe {
        let p = &raw const FRAMEBUFFER;
        let fb = (*p).as_ref().expect("GPU not initialized");
        (fb.ptr, fb.len, fb.width as usize, fb.height as usize)
    };

    if x + w > fb_w || y + h > fb_h || w == 0 || h == 0 {
        return usize::MAX;
    }

    let fb_buf = unsafe { core::slice::from_raw_parts_mut(fb_ptr, fb_len) };

    // Translate user virtual address to kernel-accessible pointer
    const READABLE: VmFlags<Sv39> = build_flags("RV");
    let process = unsafe { PROCESSES.get_mut() }.get_mut(caller_entity).unwrap();
    let total_bytes = w * h * 4;

    // Copy row-by-row to handle potential page boundary crossings
    for row in 0..h {
        let row_start = data_ptr + row * w * 4;
        let row_bytes = w * 4;

        if let Some(ptr) = process.address_space.translate::<u8>(
            VAddr::new(row_start), READABLE
        ) {
            let user_row = unsafe { core::slice::from_raw_parts(ptr.as_ptr(), row_bytes) };
            for col in 0..w {
                let src_off = col * 4;
                let alpha = user_row[src_off + 3];
                if alpha == 0 {
                    continue;
                }
                let fb_off = ((y + row) * fb_w + (x + col)) * 4;
                fb_buf[fb_off] = user_row[src_off];
                fb_buf[fb_off + 1] = user_row[src_off + 1];
                fb_buf[fb_off + 2] = user_row[src_off + 2];
                fb_buf[fb_off + 3] = user_row[src_off + 3];
            }
        } else {
            return usize::MAX;
        }
    }

    let gpu = unsafe {
        let p = &raw mut GPU;
        (*p).as_mut().expect("GPU not initialized")
    };
    gpu.flush().expect("GPU flush failed");

    0
}
```

```rust
// ========== PRNG ==========

struct Rng {
    state: u64,
}

impl Rng {
    fn new(seed: u64) -> Self {
        Self { state: if seed == 0 { 1 } else { seed } }
    }

    fn next(&mut self) -> u64 {
        self.state ^= self.state << 13;
        self.state ^= self.state >> 7;
        self.state ^= self.state << 17;
        self.state
    }

    fn next_piece(&mut self) -> u8 {
        (self.next() % 7) as u8
    }
}

// ========== Game State ==========

#[derive(Clone, Copy, PartialEq)]
enum GameState {
    Playing,
    GameOver,
}

struct Piece {
    shape: u8,    // 0-6
    rotation: u8, // 0-3
    row: i8,
    col: i8,
}

impl Piece {
    fn cells(&self) -> [(i8, i8); 4] {
        let offsets = PIECES[self.shape as usize][self.rotation as usize];
        [
            (self.row + offsets[0].0, self.col + offsets[0].1),
            (self.row + offsets[1].0, self.col + offsets[1].1),
            (self.row + offsets[2].0, self.col + offsets[2].1),
            (self.row + offsets[3].0, self.col + offsets[3].1),
        ]
    }

    fn rotated(&self) -> Self {
        Self {
            shape: self.shape,
            rotation: (self.rotation + 1) % 4,
            row: self.row,
            col: self.col,
        }
    }

    fn moved(&self, dr: i8, dc: i8) -> Self {
        Self {
            shape: self.shape,
            rotation: self.rotation,
            row: self.row + dr,
            col: self.col + dc,
        }
    }
}

struct Game {
    board: [u8; BOARD_W * BOARD_H], // 0 = empty, 1-7 = piece color
    current: Piece,
    next_shape: u8,
    score: u32,
    level: u32,
    lines: u32,
    state: GameState,
    rng: Rng,
    tick_ms: usize,
    soft_drop: bool,
}

impl Game {
    fn new(seed: u64) -> Self {
        let mut rng = Rng::new(seed);
        let current_shape = rng.next_piece();
        let next_shape = rng.next_piece();
        Self {
            board: [0; BOARD_W * BOARD_H],
            current: Piece { shape: current_shape, rotation: 0, row: 0, col: 4 },
            next_shape,
            score: 0,
            level: 1,
            lines: 0,
            state: GameState::Playing,
            rng,
            tick_ms: 800,
            soft_drop: false,
        }
    }

    fn fits(&self, piece: &Piece) -> bool {
        for (r, c) in piece.cells() {
            if r < 0 || r >= BOARD_H as i8 || c < 0 || c >= BOARD_W as i8 {
                return false;
            }
            if self.board[r as usize * BOARD_W + c as usize] != 0 {
                return false;
            }
        }
        true
    }

    fn ghost_row(&self) -> i8 {
        let mut ghost = Piece { ..*&self.current };
        while self.fits(&ghost.moved(1, 0)) {
            ghost.row += 1;
        }
        ghost.row
    }

    fn lock_piece(&mut self) {
        let color = self.current.shape + 1;
        for (r, c) in self.current.cells() {
            if r >= 0 && r < BOARD_H as i8 && c >= 0 && c < BOARD_W as i8 {
                self.board[r as usize * BOARD_W + c as usize] = color;
            }
        }
    }

    fn clear_lines(&mut self) -> u32 {
        let mut cleared = 0u32;
        let mut dst = BOARD_H as i32 - 1;

        for src in (0..BOARD_H).rev() {
            let full = (0..BOARD_W).all(|c| self.board[src * BOARD_W + c] != 0);
            if full {
                cleared += 1;
            } else {
                if dst as usize != src {
                    for c in 0..BOARD_W {
                        self.board[dst as usize * BOARD_W + c] = self.board[src * BOARD_W + c];
                    }
                }
                dst -= 1;
            }
        }

        // Clear remaining top rows
        for r in 0..=dst {
            for c in 0..BOARD_W {
                self.board[r as usize * BOARD_W + c] = 0;
            }
        }

        cleared
    }

    fn add_score(&mut self, lines_cleared: u32) {
        let points = match lines_cleared {
            1 => 100,
            2 => 300,
            3 => 500,
            4 => 800,
            _ => 0,
        };
        self.score += points * self.level;
        self.lines += lines_cleared;
        self.level = 1 + self.lines / 10;
        // Speed: 800ms at level 1, decreasing, minimum 100ms
        self.tick_ms = if self.level >= 8 { 100 } else { 800 - (self.level as usize - 1) * 100 };
    }

    fn spawn_next(&mut self) -> bool {
        self.current = Piece {
            shape: self.next_shape,
            rotation: 0,
            row: 0,
            col: 4,
        };
        self.next_shape = self.rng.next_piece();
        self.fits(&self.current)
    }

    fn try_move(&mut self, dr: i8, dc: i8) -> bool {
        let moved = self.current.moved(dr, dc);
        if self.fits(&moved) {
            self.current = moved;
            true
        } else {
            false
        }
    }

    fn try_rotate(&mut self) -> bool {
        let rotated = self.current.rotated();
        if self.fits(&rotated) {
            self.current = rotated;
            true
        } else {
            false
        }
    }

    fn hard_drop(&mut self) {
        while self.fits(&self.current.moved(1, 0)) {
            self.current.row += 1;
        }
    }
}

// ========== Rendering ==========

// Static buffer for cell rendering (CELL x CELL x 4 bytes)
// CELL=30, so 30*30*4 = 3600 bytes — safe for stack
fn draw_rect(x: usize, y: usize, w: usize, h: usize, color: [u8; 4]) {
    // Use a row buffer to avoid large stack allocation
    let mut row_buf = [0u8; 1280 * 4]; // max row width
    let row_bytes = w * 4;
    for i in 0..w {
        row_buf[i * 4] = color[0];
        row_buf[i * 4 + 1] = color[1];
        row_buf[i * 4 + 2] = color[2];
        row_buf[i * 4 + 3] = color[3];
    }
    for row in 0..h {
        fb_write(x as u32, (y + row) as u32, w as u32, 1, row_buf.as_ptr());
    }
}

fn draw_cell(x: usize, y: usize, color: [u8; 4]) {
    let mut buf = [0u8; CELL * CELL * 4];
    for r in 0..CELL {
        for c in 0..CELL {
            let idx = (r * CELL + c) * 4;
            // Border effect: 1px dark border
            if r == 0 || c == 0 || r == CELL - 1 || c == CELL - 1 {
                let dim = |v: u8| (v as u16 * 60 / 100) as u8;
                buf[idx] = dim(color[0]);
                buf[idx + 1] = dim(color[1]);
                buf[idx + 2] = dim(color[2]);
                buf[idx + 3] = color[3];
            } else {
                buf[idx] = color[0];
                buf[idx + 1] = color[1];
                buf[idx + 2] = color[2];
                buf[idx + 3] = color[3];
            }
        }
    }
    fb_write(x as u32, y as u32, CELL as u32, CELL as u32, buf.as_ptr());
}

fn draw_ghost_cell(x: usize, y: usize, color: [u8; 4]) {
    let mut buf = [0u8; CELL * CELL * 4];
    for r in 0..CELL {
        for c in 0..CELL {
            let idx = (r * CELL + c) * 4;
            // Ghost: only draw border outline
            if r <= 1 || c <= 1 || r >= CELL - 2 || c >= CELL - 2 {
                buf[idx] = color[0];
                buf[idx + 1] = color[1];
                buf[idx + 2] = color[2];
                buf[idx + 3] = GHOST_ALPHA;
            }
            // Interior is transparent (alpha=0), so board shows through
        }
    }
    fb_write(x as u32, y as u32, CELL as u32, CELL as u32, buf.as_ptr());
}

fn draw_char(x: usize, y: usize, ch: u8, color: [u8; 4]) {
    let glyph = GLYPHS.iter().find(|(c, _)| *c == ch);
    let rows = match glyph {
        Some((_, rows)) => rows,
        None => return,
    };
    let w = GLYPH_W * FONT_SCALE;
    let h = GLYPH_H * FONT_SCALE;
    let mut buf = [0u8; (5 * 3) * (7 * 3) * 4]; // 15x21 * 4
    for gr in 0..GLYPH_H {
        for gc in 0..GLYPH_W {
            let on = (rows[gr] >> (GLYPH_W - 1 - gc)) & 1 == 1;
            if on {
                for sr in 0..FONT_SCALE {
                    for sc in 0..FONT_SCALE {
                        let pr = gr * FONT_SCALE + sr;
                        let pc = gc * FONT_SCALE + sc;
                        let idx = (pr * w + pc) * 4;
                        buf[idx] = color[0];
                        buf[idx + 1] = color[1];
                        buf[idx + 2] = color[2];
                        buf[idx + 3] = color[3];
                    }
                }
            }
        }
    }
    fb_write(x as u32, y as u32, w as u32, h as u32, buf.as_ptr());
}

fn draw_string(x: usize, y: usize, s: &[u8], color: [u8; 4]) {
    for (i, &ch) in s.iter().enumerate() {
        draw_char(x + i * (GLYPH_W * FONT_SCALE + 2), y, ch, color);
    }
}

fn draw_number(x: usize, y: usize, mut n: u32, color: [u8; 4]) {
    let mut digits = [0u8; 10];
    let mut len = 0;
    if n == 0 {
        digits[0] = b'0';
        len = 1;
    } else {
        while n > 0 {
            digits[len] = b'0' + (n % 10) as u8;
            n /= 10;
            len += 1;
        }
        // Reverse
        for i in 0..len / 2 {
            digits.swap(i, len - 1 - i);
        }
    }
    draw_string(x, y, &digits[..len], color);
}

fn draw_falling_piece(game: &Game) {
    // Draw ghost piece
    let ghost_row = game.ghost_row();
    let ghost = Piece {
        shape: game.current.shape,
        rotation: game.current.rotation,
        row: ghost_row,
        col: game.current.col,
    };
    for (r, c) in ghost.cells() {
        if r >= 0 && r < BOARD_H as i8 && c >= 0 && c < BOARD_W as i8 {
            draw_ghost_piece_cell(r as usize, c as usize, game.current.shape);
        }
    }

    // Draw current piece
    for (r, c) in game.current.cells() {
        if r >= 0 && r < BOARD_H as i8 && c >= 0 && c < BOARD_W as i8 {
            draw_piece_cell(r as usize, c as usize, game.current.shape);
        }
    }
}

fn erase_falling_piece(game: &Game) {
    // Erase ghost
    let ghost_row = game.ghost_row();
    let ghost = Piece {
        shape: game.current.shape,
        rotation: game.current.rotation,
        row: ghost_row,
        col: game.current.col,
    };
    for (r, c) in ghost.cells() {
        if r >= 0 && r < BOARD_H as i8 && c >= 0 && c < BOARD_W as i8 {
            draw_board_cell(r as usize, c as usize);
        }
    }

    // Erase current piece
    for (r, c) in game.current.cells() {
        if r >= 0 && r < BOARD_H as i8 && c >= 0 && c < BOARD_W as i8 {
            draw_board_cell(r as usize, c as usize);
        }
    }
}

fn redraw_board(game: &Game) {
    for r in 0..BOARD_H {
        for c in 0..BOARD_W {
            let val = game.board[r * BOARD_W + c];
            if val == 0 {
                draw_board_cell(r, c);
            } else {
                draw_piece_cell(r, c, val - 1);
            }
        }
    }
}

    let ov_x = BOARD_X + 10;
    let ov_y = BOARD_Y + (BOARD_H * CELL - ov_h) / 2;
    draw_rect(ov_x, ov_y, ov_w, ov_h, [0x00, 0x00, 0x00, 0xD0]);

    let text_x = ov_x + 30;
    let text_y = ov_y + 20;
    draw_string(text_x, text_y, b"OVER", [0x00, 0x40, 0xFF, 0xFF]);
    draw_string(text_x, text_y + 40, b"ENTER", [0xCC, 0xCC, 0xCC, 0xFF]);
}

// ========== Input ==========

fn poll_key() -> Option<u8> {
    let mut buf = [0u8; 1];
    let ret = read(STDIN, &mut buf);
    if ret > 0 && buf[0] != 0 {
        Some(buf[0])
    } else {
        None
    }
}

fn drain_input(game: &mut Game) -> bool {
    let mut moved = false;
    loop {
        match poll_key() {
            Some(b'a') => {
                erase_falling_piece(game);
                game.try_move(0, -1);
                draw_falling_piece(game);
                moved = true;
            }
            Some(b'd') => {
                erase_falling_piece(game);
                game.try_move(0, 1);
                draw_falling_piece(game);
                moved = true;
            }
            Some(b'w') => {
                erase_falling_piece(game);
                game.try_rotate();
                draw_falling_piece(game);
                moved = true;
            }
            Some(b's') => {
                game.soft_drop = true;
                moved = true;
            }
            Some(b' ') => {
                erase_falling_piece(game);
                game.hard_drop();
                draw_falling_piece(game);
                moved = true;
                // Immediately lock after hard drop
                return true; // signal hard drop
            }
            Some(_) => {}
            None => break,
        }
    }
    false // no hard drop
}

fn wait_for_key() {
    loop {
        let mut buf = [0u8; 1];
        let ret = read(STDIN, &mut buf);
        if ret > 0 && buf[0] != 0 {
            return;
        }
        sched_yield();
        sleep(50);
    }
}

// ========== Main Game Loop ==========

/// Entry point for the Tetris game.
pub fn run_game() {
    let (_fb_w, _fb_h) = fb_info();

    loop {
        let seed = get_time() as u64;
        let mut game = Game::new(seed);

        draw_initial_screen(&game);

        let mut last_tick = get_time();

        loop {
            // Process input
            let hard_dropped = drain_input(&mut game);

            if hard_dropped || game.state == GameState::GameOver {
                // fall through to tick logic for hard drop
            }

            let now = get_time();
            let tick_interval = if game.soft_drop { 50 } else { game.tick_ms };
            let should_tick = hard_dropped || (now - last_tick) as usize >= tick_interval;

            if should_tick && game.state == GameState::Playing {
                last_tick = now;
                game.soft_drop = false;

                erase_falling_piece(&game);

                if hard_dropped || !game.try_move(1, 0) {
                    // Lock piece
                    game.lock_piece();
                    let cleared = game.clear_lines();
                    if cleared > 0 {
                        game.add_score(cleared);
                        redraw_board(&game);
                        draw_panel_values(&game);
                    }

                    // Spawn next
                    if !game.spawn_next() {
                        game.state = GameState::GameOver;
                        redraw_board(&game);
                        draw_game_over();
                    } else {
                        draw_next_piece(game.next_shape);
                    }
                }

                if game.state == GameState::Playing {
                    draw_falling_piece(&game);
                }
            }

            if game.state == GameState::GameOver {
                // Wait for any key to restart
                wait_for_key();
                break; // restart outer loop
            }

            sched_yield();
            sleep(16); // ~60fps frame pacing
        }
    }
}
```
