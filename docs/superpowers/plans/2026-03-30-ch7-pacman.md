```rust
//! DMA bump allocator and VirtIO Hal implementation.
//!
//! Uses a fixed region at 0x8500_0000 for DMA buffers (above kernel heap).
//! The kernel identity-maps physical RAM, so phys_to_virt is identity.

use core::ptr::NonNull;
use core::sync::atomic::{AtomicUsize, Ordering};
use virtio_drivers::{BufferDirection, Hal, PhysAddr};

/// Fixed base address for the DMA pool — above kernel heap, within QEMU RAM.
const POOL_BASE: usize = 0x8500_0000;

/// 5 MiB pool for VirtIO GPU framebuffer + virtqueues.
const POOL_SIZE: usize = 5 * 1024 * 1024;

const PAGE_SIZE: usize = 4096;

/// Atomic offset into the pool.
static NEXT: AtomicUsize = AtomicUsize::new(0);

/// Bump-allocate `size` bytes with the given alignment from the pool.
fn bump_alloc(size: usize, align: usize) -> *mut u8 {
    loop {
        let current = NEXT.load(Ordering::Relaxed);
        let addr = POOL_BASE + current;
        let aligned = (addr + align - 1) & !(align - 1);
        let offset = aligned - POOL_BASE;
        let new_offset = offset + size;
        if new_offset > POOL_SIZE {
            return core::ptr::null_mut();
        }
        if NEXT
            .compare_exchange(current, new_offset, Ordering::Relaxed, Ordering::Relaxed)
            .is_ok()
        {
            unsafe { core::ptr::write_bytes(aligned as *mut u8, 0, size) };
            return aligned as *mut u8;
        }
    }
}

/// VirtIO HAL: identity-mapped, DMA from fixed pool.
pub struct HalImpl;

impl Hal for HalImpl {
    fn dma_alloc(pages: usize, _direction: BufferDirection) -> (PhysAddr, NonNull<u8>) {
        let size = pages * PAGE_SIZE;
        let ptr = bump_alloc(size, PAGE_SIZE);
        assert!(!ptr.is_null(), "DMA allocation failed");
        (ptr as PhysAddr, NonNull::new(ptr).unwrap())
    }

    fn dma_dealloc(_paddr: PhysAddr, _vaddr: NonNull<u8>, _pages: usize) -> i32 {
        0
    }

    fn mmio_phys_to_virt(paddr: PhysAddr, _size: usize) -> NonNull<u8> {
        NonNull::new(paddr as *mut u8).unwrap()
    }

    fn share(buffer: NonNull<[u8]>, _direction: BufferDirection) -> PhysAddr {
        buffer.as_ptr() as *mut u8 as PhysAddr
    }

    fn unshare(_paddr: PhysAddr, _buffer: NonNull<[u8]>, _direction: BufferDirection) {}
}
```

```rust
//! VirtIO device initialization: GPU framebuffer and keyboard input.

use crate::allocator::HalImpl;
use core::ptr::NonNull;
use virtio_drivers::{
    device::{gpu::VirtIOGpu, input::VirtIOInput},
    transport::{
        mmio::{MmioTransport, VirtIOHeader},
        DeviceType, Transport,
    },
};

const VIRTIO_MMIO_BASE: usize = 0x10001000;
const VIRTIO_MMIO_END: usize = 0x10008000;
const VIRTIO_MMIO_STRIDE: usize = 0x1000;

/// Framebuffer descriptor.
pub struct Framebuffer {
    /// Pointer to framebuffer memory.
    pub ptr: *mut u8,
    /// Length of framebuffer in bytes.
    pub len: usize,
    /// Display width in pixels.
    pub width: u32,
    /// Display height in pixels.
    pub height: u32,
}

/// Result of scanning all VirtIO MMIO slots.
pub struct VirtIODevices {
    /// GPU driver and framebuffer info.
    pub gpu: (VirtIOGpu<'static, HalImpl, MmioTransport>, Framebuffer),
    /// Keyboard driver (if found).
    pub keyboard: Option<VirtIOInput<HalImpl, MmioTransport>>,
}

/// Scan all VirtIO MMIO slots, initialize GPU and keyboard devices.
pub fn init() -> VirtIODevices {
    let mut gpu_transport: Option<MmioTransport> = None;
    let mut kbd_transport: Option<MmioTransport> = None;

    let mut addr = VIRTIO_MMIO_BASE;
    while addr <= VIRTIO_MMIO_END {
        let header = NonNull::new(addr as *mut VirtIOHeader).unwrap();
        if let Ok(transport) = unsafe { MmioTransport::new(header) } {
            match transport.device_type() {
                DeviceType::GPU => {
                    tg_console::log::info!("found VirtIO GPU at {addr:#x}");
                    gpu_transport = Some(transport);
                }
                DeviceType::Input => {
                    tg_console::log::info!("found VirtIO Input at {addr:#x}");
                    kbd_transport = Some(transport);
                }
                other => {
                    tg_console::log::debug!("skipping VirtIO device type {other:?} at {addr:#x}");
                }
            }
        }
        addr += VIRTIO_MMIO_STRIDE;
    }

    let gpu_transport = gpu_transport.expect("no VirtIO GPU device found");
    let mut gpu = VirtIOGpu::new(gpu_transport).expect("failed to create VirtIOGpu");
    let (width, height) = gpu.resolution().expect("failed to get resolution");
    let (ptr, len) = {
        let buf = gpu.setup_framebuffer().expect("failed to setup framebuffer");
        (buf.as_mut_ptr(), buf.len())
    };

    let fb = Framebuffer { ptr, len, width, height };
    let keyboard = kbd_transport.map(|t| VirtIOInput::new(t).expect("failed to create VirtIOInput"));
    if keyboard.is_none() {
        tg_console::log::warn!("no VirtIO keyboard device found");
    }

    VirtIODevices { gpu: (gpu, fb), keyboard }
}
```

```rust
/// Translate a VirtIO keycode to a byte.
/// Returns keycode for press events, keycode|0x80 for release events.
fn translate_keycode(code: u16) -> u8 {
    match code {
        17 => 17,   // W
        30 => 30,   // A
        31 => 31,   // S
        32 => 32,   // D
        103 => 103, // Up
        105 => 105, // Left
        106 => 106, // Right
        108 => 108, // Down
        57 => 57,   // Space
        28 => 28,   // Enter
        63 => 63,   // F5
        _ => code as u8,
    }
}

/// Poll VirtIO keyboard for the next key event.
/// Returns keycode for press, keycode|0x80 for release, drains sync events.
#[cfg(target_arch = "riscv64")]
fn keyboard_trygetchar() -> Option<u8> {
    let kbd = unsafe {
        let p = &raw mut KEYBOARD;
        (*p).as_mut()
    }?;
    loop {
        match kbd.pop_pending_event() {
            Some(event) => {
                // EV_KEY = 1
                if event.event_type == 1 {
                    let keycode = translate_keycode(event.code);
                    if event.value == 1 {
                        // Key press
                        return Some(keycode);
                    } else if event.value == 0 {
                        // Key release
                        return Some(keycode | 0x80);
                    }
                }
                // EV_SYN and other event types: drain and continue
            }
            None => return None,
        }
    }
}

/// FB_INFO: returns (width << 32) | height.
#[cfg(target_arch = "riscv64")]
fn handle_fb_info() -> usize {
    let (width, height) = unsafe {
        let p = &raw const FRAMEBUFFER;
        let fb = (*p).as_ref().expect("GPU not initialized");
        (fb.width, fb.height)
    };
    ((width as usize) << 32) | (height as usize)
}

/// FB_WRITE: copy BGRA pixels from user buffer into framebuffer (no flush).
#[cfg(target_arch = "riscv64")]
fn handle_fb_write(x: usize, y: usize, w: usize, h: usize, data_ptr: usize) -> usize {
    use tg_kernel_vm::page_table::VmFlags;

    let (fb_ptr, fb_len, fb_w, fb_h) = unsafe {
        let p = &raw const FRAMEBUFFER;
        let fb = (*p).as_ref().expect("GPU not initialized");
        (fb.ptr, fb.len, fb.width as usize, fb.height as usize)
    };

    if x + w > fb_w || y + h > fb_h || w == 0 || h == 0 {
        return usize::MAX;
    }

    let fb_buf = unsafe { core::slice::from_raw_parts_mut(fb_ptr, fb_len) };

    const READABLE: VmFlags<Sv39> = build_flags("RV");
    let current = PROCESSOR.get_mut().current().unwrap();

    for row in 0..h {
        let row_va = data_ptr + row * w * 4;
        if let Some(ptr) = current.address_space.translate::<u8>(VAddr::new(row_va), READABLE) {
            let user_row = unsafe { core::slice::from_raw_parts(ptr.as_ptr(), w * 4) };
            for col in 0..w {
                let src_off = col * 4;
                let alpha = user_row[src_off + 3];
                if alpha == 0 {
                    continue;
                }
                let fb_off = ((y + row) * fb_w + (x + col)) * 4;
                fb_buf[fb_off] = user_row[src_off];
                fb_buf[fb_off + 1] = user_row[src_off + 1];
                fb_buf[fb_off + 2] = user_row[src_off + 2];
                fb_buf[fb_off + 3] = user_row[src_off + 3];
            }
        } else {
            return usize::MAX;
        }
    }

    0
}

/// FB_FLUSH: flush the GPU framebuffer to the display.
#[cfg(target_arch = "riscv64")]
fn handle_fb_flush() -> usize {
    let gpu = unsafe {
        let p = &raw mut GPU;
        (*p).as_mut().expect("GPU not initialized")
    };
    gpu.flush().expect("GPU flush failed");
    0
}
```

```rust
/// Fill a rectangle with a solid color. Uses a 16x16 tile to limit stack usage.
fn fill_rect(x: usize, y: usize, w: usize, h: usize, color: [u8; 4]) {
    const TILE: usize = 16;
    let mut buf = [0u8; TILE * TILE * 4];
    for i in 0..(TILE * TILE) {
        buf[i * 4] = color[0];
        buf[i * 4 + 1] = color[1];
        buf[i * 4 + 2] = color[2];
        buf[i * 4 + 3] = color[3];
    }

    let mut ry = 0;
    while ry < h {
        let th = if ry + TILE > h { h - ry } else { TILE };
        let mut rx = 0;
        while rx < w {
            let tw = if rx + TILE > w { w - rx } else { TILE };
            fb_write((x + rx) as u32, (y + ry) as u32, tw as u32, th as u32, buf.as_ptr());
            rx += TILE;
        }
        ry += TILE;
    }
}

/// Draw a single maze cell at grid position (cx, cy).
fn draw_wall_cell(cx: usize, cy: usize, maze: &[u8]) {
    let px = MAZE_X + cx * CELL;
    let py = MAZE_Y + cy * CELL;

    // Fill with shadow base
    fill_rect(px, py, CELL, CELL, WALL_SHADOW);

    // Check neighbors for edge highlighting
    let has_path = |dx: i32, dy: i32| -> bool {
        let nx = cx as i32 + dx;
        let ny = cy as i32 + dy;
        if nx < 0 || nx >= MAZE_W as i32 || ny < 0 || ny >= MAZE_H as i32 {
            return true; // Outside maze = path
        }
        let c = maze[ny as usize * MAZE_W + nx as usize];
        c != W
    };

    // Edge highlights (2px strips)
    if has_path(0, -1) { fill_rect(px, py, CELL, 2, WALL_EDGE); }
    if has_path(0, 1) { fill_rect(px, py + CELL - 2, CELL, 2, WALL_EDGE); }
    if has_path(-1, 0) { fill_rect(px, py, 2, CELL, WALL_EDGE); }
    if has_path(1, 0) { fill_rect(px + CELL - 2, py, 2, CELL, WALL_EDGE); }

    // Interior fill
    fill_rect(px + 2, py + 2, CELL - 4, CELL - 4, WALL_FACE);
}

/// Draw a path cell (black with optional dot/pellet).
fn draw_path_cell(cx: usize, cy: usize, cell_type: u8, pellet_visible: bool) {
    let px = MAZE_X + cx * CELL;
    let py = MAZE_Y + cy * CELL;
    fill_rect(px, py, CELL, CELL, PATH_COLOR);

    match cell_type {
        D => {
            // Dot: 4x4 at cell center
            fill_rect(px + 12, py + 12, 4, 4, DOT_COLOR);
        }
        P => {
            if pellet_visible {
                // Power pellet: 14x14 glow + 10x10 bright center
                fill_rect(px + 7, py + 7, 14, 14, PELLET_GLOW);
                fill_rect(px + 9, py + 9, 10, 10, PELLET_COLOR);
            }
        }
        _ => {}
    }
}

/// Draw the entire maze.
fn draw_maze(maze: &[u8], pellet_visible: bool) {
    for cy in 0..MAZE_H {
        for cx in 0..MAZE_W {
            let cell = maze[cy * MAZE_W + cx];
            match cell {
                W => draw_wall_cell(cx, cy, maze),
                G => {
                    // Ghost house: draw as path with gate color for top edges
                    let px = MAZE_X + cx * CELL;
                    let py = MAZE_Y + cy * CELL;
                    fill_rect(px, py, CELL, CELL, PATH_COLOR);
                    if cy == 9 { fill_rect(px, py, CELL, 2, GATE_COLOR); }
                }
                _ => draw_path_cell(cx, cy, cell, pellet_visible),
            }
        }
    }
}

/// Draw a ghost at grid position with given color.
fn draw_ghost(cx: u8, cy: u8, ghost_idx: usize, dir: u8, frightened: bool, flash: bool, eaten: bool) {
    let px = MAZE_X + cx as usize * CELL + 4;
    let py = MAZE_Y + cy as usize * CELL + 4;

    if eaten {
        // Eyes only on black background
        fill_rect(px, py, 20, 20, PATH_COLOR);
        // White eyes
        fill_rect(px + 4, py + 4, 6, 6, GHOST_EYE_W);
        fill_rect(px + 12, py + 4, 6, 6, GHOST_EYE_W);
        // Pupils offset toward movement direction
        let (pdx, pdy): (usize, usize) = match dir {
            DIR_RIGHT => (3, 1),
            DIR_LEFT => (0, 1),
            DIR_UP => (1, 0),
            _ => (1, 3),
        };
        fill_rect(px + 4 + pdx, py + 4 + pdy, 3, 3, GHOST_EYE_B);
        fill_rect(px + 12 + pdx, py + 4 + pdy, 3, 3, GHOST_EYE_B);
        return;
    }

    if frightened {
        let body = if flash { FRIGHT_FLASH } else { FRIGHT_BODY };
        fill_rect(px, py, 20, 20, body);
        // Frightened face: two small eye dots
        fill_rect(px + 5, py + 6, 4, 4, FRIGHT_FACE);
        fill_rect(px + 13, py + 6, 4, 4, FRIGHT_FACE);
        // Wavy mouth: 6 small dots in zigzag
        for i in 0..6 {
            let mx = px + 2 + i * 3;
            let my = if i % 2 == 0 { py + 14 } else { py + 16 };
            fill_rect(mx, my, 2, 2, FRIGHT_FACE);
        }
        return;
    }

    // Normal ghost
    let body = GHOST_COLORS[ghost_idx];
    let highlight = GHOST_HIGHLIGHTS[ghost_idx];
    fill_rect(px, py, 20, 20, body);
    fill_rect(px + 4, py + 0, 12, 12, highlight);

    // White eyes
    fill_rect(px + 4, py + 4, 6, 6, GHOST_EYE_W);
    fill_rect(px + 12, py + 4, 6, 6, GHOST_EYE_W);

    // Pupils offset toward movement direction
    let (pdx, pdy): (usize, usize) = match dir {
        DIR_RIGHT => (3, 1),
        DIR_LEFT => (0, 1),
        DIR_UP => (1, 0),
        _ => (1, 3),
    };
    fill_rect(px + 4 + pdx, py + 4 + pdy, 3, 3, GHOST_EYE_B);
    fill_rect(px + 12 + pdx, py + 4 + pdy, 3, 3, GHOST_EYE_B);

    // Ghost skirt: alternating strips at bottom
    for i in 0..5 {
        let sx = px + i * 4;
        let color = if i % 2 == 0 { body } else { WALL_SHADOW };
        fill_rect(sx, py + 18, 4, 2, color);
    }
}

/// Draw a text string at pixel position (x, y) with given color.
fn draw_string(x: usize, y: usize, s: &[u8], color: [u8; 4]) {
    for (i, &ch) in s.iter().enumerate() {
        if let Some(bits) = glyph(ch) {
            let cx = x + i * (CHAR_W + 1) * FONT_SCALE;
            let mut buf = [0u8; CHAR_W * FONT_SCALE * CHAR_H * FONT_SCALE * 4];
            let bw = CHAR_W * FONT_SCALE;
            for row in 0..CHAR_H {
                for col in 0..CHAR_W {
                    if bits[row] & (1 << (CHAR_W - 1 - col)) != 0 {
                        for sy in 0..FONT_SCALE {
                            for sx in 0..FONT_SCALE {
                                let px = col * FONT_SCALE + sx;
                                let py = row * FONT_SCALE + sy;
                                let off = (py * bw + px) * 4;
                                buf[off] = color[0];
                                buf[off + 1] = color[1];
                                buf[off + 2] = color[2];
                                buf[off + 3] = color[3];
                            }
                        }
                    }
                }
            }
            fb_write(cx as u32, y as u32, bw as u32, (CHAR_H * FONT_SCALE) as u32, buf.as_ptr());
        }
    }
}

/// Draw a number at pixel position.
fn draw_number(x: usize, y: usize, mut val: u32, color: [u8; 4]) {
    let mut digits = [0u8; 10];
    let mut len = 0;
    if val == 0 {
        digits[0] = b'0';
        len = 1;
    } else {
        while val > 0 {
            digits[len] = b'0' + (val % 10) as u8;
            val /= 10;
            len += 1;
        }
        // Reverse
        let mut i = 0;
        let mut j = len - 1;
        while i < j {
            digits.swap(i, j);
            i += 1;
            j -= 1;
        }
    }
    draw_string(x, y, &digits[..len], color);
}

// ═══════════════════════════════════════════════════════════════
// Maze helpers
// ═══════════════════════════════════════════════════════════════

fn is_walkable(maze: &[u8], x: i32, y: i32) -> bool {
    if x < 0 || x >= MAZE_W as i32 || y < 0 || y >= MAZE_H as i32 {
        return false;
    }
    let cell = maze[y as usize * MAZE_W + x as usize];
    cell != W && cell != G
}

fn is_tunnel_row(y: u8) -> bool {
    y == 8 || y == 10 || y == 12
}

fn wrap_x(x: i32) -> u8 {
    if x < 0 { (MAZE_W as i32 - 1) as u8 }
    else if x >= MAZE_W as i32 { 0 }
    else { x as u8 }
}

fn manhattan(x1: u8, y1: u8, x2: u8, y2: u8) -> u32 {
    let dx = if x1 > x2 { x1 - x2 } else { x2 - x1 } as u32;
    let dy = if y1 > y2 { y1 - y2 } else { y2 - y1 } as u32;
    dx + dy
}

fn dir_delta(dir: u8) -> (i32, i32) {
    match dir {
        DIR_UP => (0, -1),
        DIR_RIGHT => (1, 0),
        DIR_DOWN => (0, 1),
        DIR_LEFT => (-1, 0),
        _ => (0, 0),
    }
}

fn opposite_dir(dir: u8) -> u8 {
    match dir {
        DIR_UP => DIR_DOWN,
        DIR_DOWN => DIR_UP,
        DIR_LEFT => DIR_RIGHT,
        DIR_RIGHT => DIR_LEFT,
        _ => DIR_NONE,
    }
}

/// Count remaining dots + power pellets in the maze.
fn count_dots(maze: &[u8]) -> u32 {
    let mut count = 0u32;
    for &cell in maze.iter() {
        if cell == D || cell == P { count += 1; }
    }
    count
}

// ═══════════════════════════════════════════════════════════════
// Simple PRNG (xorshift32)
// ═══════════════════════════════════════════════════════════════

fn xorshift32(state: &mut u32) -> u32 {
    let mut x = *state;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    *state = x;
    x
}

// ═══════════════════════════════════════════════════════════════
// High score persistence
// ═══════════════════════════════════════════════════════════════

fn load_high_score() -> u32 {
    let fd = open("pacman_hiscore\0", OpenFlags::RDONLY);
    if fd < 0 { return 0; }
    let mut buf = [0u8; 4];
    let n = read(fd as usize, &mut buf);
    close(fd as usize);
    if n == 4 {
        u32::from_le_bytes(buf)
    } else {
        0
    }
}

fn save_high_score(score: u32) {
    let fd = open("pacman_hiscore\0", OpenFlags::CREATE | OpenFlags::WRONLY | OpenFlags::TRUNC);
    if fd < 0 { return; }
    let buf = score.to_le_bytes();
    write(fd as usize, &buf);
    close(fd as usize);
}

// Ghost AI (child process)
// ═══════════════════════════════════════════════════════════════

/// Signal flags for ghost child processes
static mut FRIGHTENED_TICKS: u32 = 0;
static mut RESPAWNING: bool = false;

fn handle_power_pellet() {
    unsafe { FRIGHTENED_TICKS = 48; } // ~6 seconds
}

fn handle_eaten() {
    unsafe { RESPAWNING = true; }
}

/// Ghost child process main loop.
fn ghost_main(ghost_idx: usize, read_fd: usize, write_fd: usize) {
    // Register signal handlers
    let power_action = SignalAction { handler: handle_power_pellet as usize, mask: 0 };
    sigaction(SIGUSR1, &power_action, core::ptr::null_mut());
    let eaten_action = SignalAction { handler: handle_eaten as usize, mask: 0 };
    sigaction(SIGUSR2, &eaten_action, core::ptr::null_mut());

    let mut rng_state = (ghost_idx as u32 + 1) * 12345 + get_time() as u32;

    // Scatter/chase timing: scatter starts first
    let mut mode_tick: u32 = 0;
    let mut scatter = true;
    let mut current_dir: u8 = DIR_UP;

    loop {
        // Read 7 bytes: [pac_x, pac_y, pac_dir, ghost_x, ghost_y, blinky_x, blinky_y]
        let mut input = [0u8; 7];
        let n = pipe_read(read_fd, &mut input);
        if n <= 0 { break; } // Parent closed pipe

        let pac_x = input[0];
        let pac_y = input[1];
        let pac_dir = input[2];
        let ghost_x = input[3];
        let ghost_y = input[4];
        let _blinky_x = input[5];
        let _blinky_y = input[6];

        let frightened = unsafe { FRIGHTENED_TICKS > 0 };
        if frightened {
            unsafe { FRIGHTENED_TICKS -= 1; }
        }

        let respawning = unsafe { RESPAWNING };
        if respawning {
            // Target ghost house center
            let home_x = GHOST_START[ghost_idx].0;
            let home_y = GHOST_START[ghost_idx].1;
            if ghost_x == home_x && ghost_y == home_y {
                unsafe { RESPAWNING = false; }
            }
            // Move toward home
            let chosen = pick_best_direction(ghost_x, ghost_y, home_x, home_y, current_dir, true);
            current_dir = chosen;
            pipe_write(write_fd, &[chosen]);
            mode_tick += 1;
            continue;
        }

        // Determine target based on mode
        mode_tick += 1;

        // Scatter/chase cycling: 56 ticks scatter (7s), 160 ticks chase (20s)
        let cycle_tick = mode_tick % (56 + 160);
        scatter = cycle_tick < 56;

        let (target_x, target_y) = if frightened {
            // Random target
            let rx = (xorshift32(&mut rng_state) % MAZE_W as u32) as u8;
            let ry = (xorshift32(&mut rng_state) % MAZE_H as u32) as u8;
            (rx, ry)
        } else if scatter {
            SCATTER_CORNERS[ghost_idx]
        } else {
            // Chase mode — unique per ghost
            match ghost_idx {
                0 => (pac_x, pac_y), // Blinky: direct chase
                1 => {
                    // Pinky: 4 ahead of pacman
                    let (dx, dy) = dir_delta(pac_dir);
                    let tx = (pac_x as i32 + dx * 4).clamp(0, MAZE_W as i32 - 1) as u8;
                    let ty = (pac_y as i32 + dy * 4).clamp(0, MAZE_H as i32 - 1) as u8;
                    (tx, ty)
                }
                2 => {
                    // Inky: alternate between chasing pac and random dot
                    if mode_tick % 2 == 0 { (pac_x, pac_y) }
                    else {
                        let rx = (xorshift32(&mut rng_state) % MAZE_W as u32) as u8;
                        let ry = (xorshift32(&mut rng_state) % MAZE_H as u32) as u8;
                        (rx, ry)
                    }
                }
                3 => {
                    // Clyde: chase if far, scatter if close
                    if manhattan(ghost_x, ghost_y, pac_x, pac_y) > 8 {
                        (pac_x, pac_y)
                    } else {
                        SCATTER_CORNERS[3]
                    }
                }
                _ => (pac_x, pac_y),
            }
        };

        let chosen = if frightened {
            // Random valid direction (not reverse)
            pick_random_direction(ghost_x, ghost_y, current_dir, &mut rng_state)
        } else {
            pick_best_direction(ghost_x, ghost_y, target_x, target_y, current_dir, false)
        };
        current_dir = chosen;
        pipe_write(write_fd, &[chosen]);
    }
}

/// Pick the direction that minimizes Manhattan distance to target.
/// Excludes reverse of current direction. Preference: up > left > down > right.
fn pick_best_direction(gx: u8, gy: u8, tx: u8, ty: u8, current_dir: u8, allow_ghost_house: bool) -> u8 {
    let reverse = opposite_dir(current_dir);
    let dirs = [DIR_UP, DIR_LEFT, DIR_DOWN, DIR_RIGHT];
    let mut best_dir = current_dir;
    let mut best_dist = u32::MAX;

    // Build a minimal maze view — ghost needs to know walkability
    // Since ghost doesn't have the maze array, we use the position info
    // For simplicity, ghost trusts parent's position updates and picks valid directions
    // The parent will clamp invalid moves, but ghost should prefer valid ones

    for &d in &dirs {
        if d == reverse && current_dir != DIR_NONE { continue; }
        let (dx, dy) = dir_delta(d);
        let nx = gx as i32 + dx;
        let ny = gy as i32 + dy;
        // Ghost can't verify walls without maze data, so just pick closest target direction
        // Parent will validate and override if needed
        if nx < 0 || nx >= MAZE_W as i32 || ny < 0 || ny >= MAZE_H as i32 { continue; }
        let dist = manhattan(nx as u8, ny as u8, tx, ty);
        if dist < best_dist {
            best_dist = dist;
            best_dir = d;
        }
    }

    best_dir
}

/// Pick a random valid direction (not reverse).
fn pick_random_direction(gx: u8, gy: u8, current_dir: u8, rng: &mut u32) -> u8 {
    let reverse = opposite_dir(current_dir);
    let dirs = [DIR_UP, DIR_RIGHT, DIR_DOWN, DIR_LEFT];
    let mut valid = [0u8; 4];
    let mut count = 0;

    for &d in &dirs {
        if d == reverse && current_dir != DIR_NONE { continue; }
        valid[count] = d;
        count += 1;
    }

    if count == 0 { return current_dir; }
    valid[(xorshift32(rng) % count as u32) as usize]
}

// Game state
// ═══════════════════════════════════════════════════════════════

struct GhostState {
    x: u8,
    y: u8,
    dir: u8,
    pid: isize,
    write_fd: usize, // parent writes to this (parent->ghost pipe write end)
    read_fd: usize,  // parent reads from this (ghost->parent pipe read end)
    frightened_ticks: u32,
    eaten: bool,
    released: bool,
    respawn_ticks: u32,
}

struct GameState {
    maze: [u8; MAZE_W * MAZE_H],
    pac_x: u8,
    pac_y: u8,
    pac_dir: u8,
    queued_dir: u8,
    score: u32,
    high_score: u32,
    lives: u32,
    level: u32,
    dots_eaten: u32,
    total_dots: u32,
    ghosts: [GhostState; NUM_GHOSTS],
    tick: u32,
    ghost_eat_combo: u32,
    pellet_visible: bool,
    rng: u32,
    paused: bool,
}

// ═══════════════════════════════════════════════════════════════
// Main game entry
// ═══════════════════════════════════════════════════════════════

pub fn run_game() {
    let _ = fb_info();

    let high_score = load_high_score();
    println!("Pac-Man starting! High score: {}", high_score);

    'outer: loop {
        // Initialize maze
        let mut maze = INITIAL_MAZE;
        let total_dots = count_dots(&maze);

        // Create pipes and fork ghost processes
        let mut ghost_states: [GhostState; NUM_GHOSTS] = core::array::from_fn(|i| GhostState {
            x: GHOST_START[i].0,
            y: GHOST_START[i].1,
            dir: DIR_UP,
            pid: 0,
            write_fd: 0,
            read_fd: 0,
            frightened_ticks: 0,
            eaten: false,
            released: i == 0, // Blinky starts released
            respawn_ticks: 0,
        });

        for i in 0..NUM_GHOSTS {
            // Parent -> Ghost pipe
            let mut p2g = [0usize; 2];
            pipe(&mut p2g);
            // Ghost -> Parent pipe
            let mut g2p = [0usize; 2];
            pipe(&mut g2p);

            let pid = fork();
            if pid == 0 {
                // Child: close unused ends
                close(p2g[1]); // Close write end of parent->ghost
                close(g2p[0]); // Close read end of ghost->parent
                // Also close other ghosts' pipes (already created ones)
                // Ghost runs its AI loop
                ghost_main(i, p2g[0], g2p[1]);
                exit(0);
                unreachable!();
            }

            // Parent: close unused ends
            close(p2g[0]); // Close read end of parent->ghost
            close(g2p[1]); // Close write end of ghost->parent

            ghost_states[i].pid = pid;
            ghost_states[i].write_fd = p2g[1];
            ghost_states[i].read_fd = g2p[0];
        }

        println!("Ghost PIDs: {} {} {} {}",
            ghost_states[0].pid, ghost_states[1].pid,
            ghost_states[2].pid, ghost_states[3].pid);

        let mut game = GameState {
            maze,
            pac_x: PAC_START_X,
            pac_y: PAC_START_Y,
            pac_dir: DIR_NONE,
            queued_dir: DIR_NONE,
            score: 0,
            high_score,
            lives: 3,
            level: 1,
            dots_eaten: 0,
            total_dots,
            ghosts: ghost_states,
            tick: 0,
            ghost_eat_combo: 0,
            pellet_visible: true,
            rng: get_time() as u32 ^ 0xDEADBEEF,
            paused: false,
        };

        // Initial draw
        fill_rect(0, 0, 1280, 800, BG_VOID);
        draw_maze(&game.maze, true);
        draw_pacman(game.pac_x, game.pac_y);
        for i in 0..NUM_GHOSTS {
            draw_ghost(game.ghosts[i].x, game.ghosts[i].y, i, DIR_UP, false, false, false);
        }
        draw_hud(game.score, game.high_score, game.level, game.lives);
        draw_string(MAZE_X + 8 * CELL, MAZE_Y + 11 * CELL, b"READY!", PELLET_COLOR);
        fb_flush();
        sleep(2000);

        // Erase READY text
        for cx in 8..14 {
            erase_cell(cx, 11, &game.maze, true);
        }
        fb_flush();

        let mut last_tick = get_time();

        // ─── Main game loop ───
        loop {
            // Poll keyboard
            let mut buf = [0u8; 1];
            loop {
                let n = read(STDIN, &mut buf);
                if n <= 0 { break; }
                let code = buf[0];
                let released = code & 0x80 != 0;
                let keycode = code & 0x7F;
                if !released {
                    match keycode {
                        KEY_W | KEY_UP => game.queued_dir = DIR_UP,
                        KEY_A | KEY_LEFT => game.queued_dir = DIR_LEFT,
                        KEY_S | KEY_DOWN => game.queued_dir = DIR_DOWN,
                        KEY_D | KEY_RIGHT => game.queued_dir = DIR_RIGHT,
                        KEY_SPACE => game.paused = !game.paused,
                        KEY_F5 => {
                            if game.score > game.high_score {
                                game.high_score = game.score;
                            }
                            save_high_score(game.high_score);
                        }
                        _ => {}
                    }
                }
            }

            if game.paused {
                sched_yield();
                sleep(50);
                continue;
            }

            let now = get_time();
            if (now - last_tick) as usize < TICK_MS { sched_yield(); continue; }
            last_tick = now;
            game.tick += 1;

            // Power pellet blink
            let new_pellet_vis = (game.tick / 4) % 2 == 0;
            if new_pellet_vis != game.pellet_visible {
                game.pellet_visible = new_pellet_vis;
                // Redraw pellet cells
                for cy in 0..MAZE_H {
                    for cx in 0..MAZE_W {
                        if game.maze[cy * MAZE_W + cx] == P {
                            draw_path_cell(cx, cy, P, game.pellet_visible);
                        }
                    }
                }
            }

            // ─── Move Pac-Man ───
            let old_px = game.pac_x;
            let old_py = game.pac_y;

            // Try queued direction first
            if game.queued_dir != DIR_NONE {
                let (dx, dy) = dir_delta(game.queued_dir);
                let nx = game.pac_x as i32 + dx;
                let ny = game.pac_y as i32 + dy;
                // Tunnel wrap
                let nx = if is_tunnel_row(game.pac_y) { wrap_x(nx) as i32 } else { nx };
                if is_walkable(&game.maze, nx, ny) {
                    game.pac_dir = game.queued_dir;
                }
            }

            // Move in current direction
            if game.pac_dir != DIR_NONE {
                let (dx, dy) = dir_delta(game.pac_dir);
                let nx = game.pac_x as i32 + dx;
                let ny = game.pac_y as i32 + dy;
                let nx = if is_tunnel_row(game.pac_y) { wrap_x(nx) as i32 } else { nx };
                if is_walkable(&game.maze, nx, ny) {
                    game.pac_x = nx as u8;
                    game.pac_y = ny as u8;
                }
            }

            // ─── Eat dot/pellet ───
            let cell_idx = game.pac_y as usize * MAZE_W + game.pac_x as usize;
            match game.maze[cell_idx] {
                D => {
                    game.maze[cell_idx] = E;
                    game.score += 10;
                    game.dots_eaten += 1;
                }
                P => {
                    game.maze[cell_idx] = E;
                    game.score += 50;
                    game.dots_eaten += 1;
                    game.ghost_eat_combo = 0;
                    // Send SIGUSR1 to all ghosts
                    for g in &mut game.ghosts {
                        if g.released && !g.eaten {
                            kill(g.pid, SIGUSR1);
                            g.frightened_ticks = 48;
                        }
                    }
                }
                _ => {}
            }

            // ─── Send state to ghosts and read their moves ───
            for i in 0..NUM_GHOSTS {
                // Release ghosts based on timing
                if !game.ghosts[i].released {
                    if game.tick >= GHOST_RELEASE_TICKS[i] ||
                       (i == 2 && game.dots_eaten >= 30) ||
                       (i == 3 && game.dots_eaten >= 80) {
                        game.ghosts[i].released = true;
                        // Move out of ghost house
                        game.ghosts[i].x = 10;
                        game.ghosts[i].y = 8;
                    }
                }

                if !game.ghosts[i].released { continue; }

                // Handle respawn countdown
                if game.ghosts[i].respawn_ticks > 0 {
                    game.ghosts[i].respawn_ticks -= 1;
                    if game.ghosts[i].respawn_ticks == 0 {
                        game.ghosts[i].eaten = false;
                        game.ghosts[i].x = GHOST_START[i].0;
                        game.ghosts[i].y = GHOST_START[i].1;
                    }
                    continue;
                }

                // Write game state to ghost
                let input = [
                    game.pac_x, game.pac_y, game.pac_dir,
                    game.ghosts[i].x, game.ghosts[i].y,
                    game.ghosts[0].x, game.ghosts[0].y, // Blinky's position
                ];
                pipe_write(game.ghosts[i].write_fd, &input[..7]);

                // Read ghost's direction choice
                let mut dir_buf = [0u8; 1];
                pipe_read(game.ghosts[i].read_fd, &mut dir_buf);
                let chosen_dir = dir_buf[0];

                // Validate and apply ghost movement
                let old_gx = game.ghosts[i].x;
                let old_gy = game.ghosts[i].y;
                let (dx, dy) = dir_delta(chosen_dir);
                let nx = game.ghosts[i].x as i32 + dx;
                let ny = game.ghosts[i].y as i32 + dy;

                // Tunnel wrap for ghosts
                let nx = if is_tunnel_row(game.ghosts[i].y) { wrap_x(nx) as i32 } else { nx };

                // Validate move (ghosts can enter ghost house if respawning)
                let can_move = if game.ghosts[i].eaten {
                    nx >= 0 && nx < MAZE_W as i32 && ny >= 0 && ny < MAZE_H as i32
                } else {
                    is_walkable(&game.maze, nx, ny)
                };

                if can_move {
                    game.ghosts[i].x = nx as u8;
                    game.ghosts[i].y = ny as u8;
                    game.ghosts[i].dir = chosen_dir;
                }

                // Frightened tick countdown (parent's view)
                if game.ghosts[i].frightened_ticks > 0 {
                    game.ghosts[i].frightened_ticks -= 1;
                }

                // Erase old ghost position
                erase_cell(old_gx, old_gy, &game.maze, game.pellet_visible);
            }

            // ─── Collision detection ───
            let mut life_lost = false;
            for i in 0..NUM_GHOSTS {
                if !game.ghosts[i].released || game.ghosts[i].eaten { continue; }
                if game.ghosts[i].respawn_ticks > 0 { continue; }

                if game.ghosts[i].x == game.pac_x && game.ghosts[i].y == game.pac_y {
                    if game.ghosts[i].frightened_ticks > 0 {
                        // Pac-Man eats ghost
                        game.ghost_eat_combo += 1;
                        let points = match game.ghost_eat_combo {
                            1 => 200,
                            2 => 400,
                            3 => 800,
                            _ => 1600,
                        };
                        game.score += points;
                        game.ghosts[i].eaten = true;
                        game.ghosts[i].respawn_ticks = 24; // 3 seconds
                        // Send SIGUSR2 to this ghost
                        kill(game.ghosts[i].pid, SIGUSR2);
                    } else {
                        // Ghost kills Pac-Man
                        life_lost = true;
                    }
                }
            }

            // ─── Render ───
            // Erase old Pac-Man, draw new
            erase_cell(old_px, old_py, &game.maze, game.pellet_visible);
            // Draw ghosts (back to front: Clyde, Inky, Pinky, Blinky)
            for &i in &[3usize, 2, 1, 0] {
                if !game.ghosts[i].released { continue; }
                if game.ghosts[i].respawn_ticks > 0 && game.ghosts[i].eaten { continue; }
                let frightened = game.ghosts[i].frightened_ticks > 0;
                let flash = frightened && game.ghosts[i].frightened_ticks < 16 && game.tick % 4 < 2;
                draw_ghost(
                    game.ghosts[i].x, game.ghosts[i].y, i,
                    game.ghosts[i].dir, frightened, flash, game.ghosts[i].eaten,
                );
            }
            draw_pacman(game.pac_x, game.pac_y);
            draw_hud(game.score, game.high_score, game.level, game.lives);
            fb_flush();

            // ─── Handle life lost ───
            if life_lost {
                game.lives -= 1;
                sleep(1000);

                if game.lives == 0 {
                    // Game over
                    if game.score > game.high_score {
                        game.high_score = game.score;
                        save_high_score(game.high_score);
                    }
                    draw_string(MAZE_X + 6 * CELL, MAZE_Y + 10 * CELL, b"GAME OVER", GHOST_COLORS[0]);
                    draw_hud(game.score, game.high_score, game.level, game.lives);
                    fb_flush();

                    // Wait for Enter to restart
                    loop {
                        let n = read(STDIN, &mut buf);
                        if n > 0 && (buf[0] & 0x7F) == KEY_ENTER && (buf[0] & 0x80) == 0 {
                            break;
                        }
                        sched_yield();
                        sleep(50);
                    }

                    // Clean up ghost processes
                    for g in &game.ghosts {
                        close(g.write_fd);
                        close(g.read_fd);
                    }
                    // Wait for ghosts to exit
                    for _ in 0..NUM_GHOSTS {
                        let mut exit_code = 0i32;
                        crate::wait(&mut exit_code);
                    }
                    continue 'outer; // Restart game
                }

                // Reset positions
                game.pac_x = PAC_START_X;
                game.pac_y = PAC_START_Y;
                game.pac_dir = DIR_NONE;
                game.queued_dir = DIR_NONE;
                for i in 0..NUM_GHOSTS {
                    game.ghosts[i].x = GHOST_START[i].0;
                    game.ghosts[i].y = GHOST_START[i].1;
                    game.ghosts[i].frightened_ticks = 0;
                    game.ghosts[i].eaten = false;
                }

                // Redraw
                draw_maze(&game.maze, game.pellet_visible);
                draw_pacman(game.pac_x, game.pac_y);
                for i in 0..NUM_GHOSTS {
                    if game.ghosts[i].released {
                        draw_ghost(game.ghosts[i].x, game.ghosts[i].y, i, DIR_UP, false, false, false);
                    }
                }
                draw_hud(game.score, game.high_score, game.level, game.lives);
                draw_string(MAZE_X + 8 * CELL, MAZE_Y + 11 * CELL, b"READY!", PELLET_COLOR);
                fb_flush();
                sleep(2000);
                erase_cell(8, 11, &game.maze, game.pellet_visible);
                erase_cell(9, 11, &game.maze, game.pellet_visible);
                erase_cell(10, 11, &game.maze, game.pellet_visible);
                erase_cell(11, 11, &game.maze, game.pellet_visible);
                erase_cell(12, 11, &game.maze, game.pellet_visible);
                erase_cell(13, 11, &game.maze, game.pellet_visible);
                fb_flush();
                last_tick = get_time();
            }

            // ─── Level complete ───
            if game.dots_eaten >= game.total_dots {
                // Flash maze 3 times
                for _ in 0..3 {
                    // Flash white
                    for cy in 0..MAZE_H {
                        for cx in 0..MAZE_W {
                            if game.maze[cy * MAZE_W + cx] == W {
                                let px = MAZE_X + cx * CELL;
                                let py = MAZE_Y + cy * CELL;
                                fill_rect(px, py, CELL, CELL, FRIGHT_FLASH);
                            }
                        }
                    }
                    fb_flush();
                    sleep(333);
                    // Flash back
                    draw_maze(&game.maze, game.pellet_visible);
                    fb_flush();
                    sleep(333);
                }

                // Next level
                game.level += 1;
                game.maze = INITIAL_MAZE;
                game.total_dots = count_dots(&game.maze);
                game.dots_eaten = 0;
                game.pac_x = PAC_START_X;
                game.pac_y = PAC_START_Y;
                game.pac_dir = DIR_NONE;
                game.queued_dir = DIR_NONE;
                game.ghost_eat_combo = 0;
                game.tick = 0;

                for i in 0..NUM_GHOSTS {
                    game.ghosts[i].x = GHOST_START[i].0;
                    game.ghosts[i].y = GHOST_START[i].1;
                    game.ghosts[i].frightened_ticks = 0;
                    game.ghosts[i].eaten = false;
                    game.ghosts[i].released = i == 0;
                    game.ghosts[i].respawn_ticks = 0;
                }

                draw_maze(&game.maze, true);
                draw_pacman(game.pac_x, game.pac_y);
                for i in 0..NUM_GHOSTS {
                    if game.ghosts[i].released {
                        draw_ghost(game.ghosts[i].x, game.ghosts[i].y, i, DIR_UP, false, false, false);
                    }
                }
                draw_hud(game.score, game.high_score, game.level, game.lives);
                draw_string(MAZE_X + 8 * CELL, MAZE_Y + 11 * CELL, b"READY!", PELLET_COLOR);
                fb_flush();
                sleep(2000);
                for cx in 8..14 {
                    erase_cell(cx, 11, &game.maze, true);
                }
                fb_flush();
                last_tick = get_time();
            }

            // Extra life at 10k
            if game.score >= 10000 && game.score - 10 < 10000 && game.lives < 5 {
                game.lives += 1;
            }
        }
    }
}
```
